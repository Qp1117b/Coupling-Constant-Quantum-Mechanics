## 2. Step 1: 嘉当矩阵组装

### 2.1 A₄ 嘉当矩阵

$$A_4 = \begin{pmatrix} 2 & -1 & 0 & 0 \\ -1 & 2 & -1 & 0 \\ 0 & -1 & 2 & -1 \\ 0 & 0 & -1 & 2 \end{pmatrix}$$

```gdscript
# cqm_matrix.gd
class_name CQMMatrix
extends RefCounted

static func A4() -> PackedFloat32Array:
    # 4×4 A₄ 嘉当矩阵（行优先）
    return PackedFloat32Array([
        2.0, -1.0,  0.0,  0.0,
       -1.0,  2.0, -1.0,  0.0,
        0.0, -1.0,  2.0, -1.0,
        0.0,  0.0, -1.0,  2.0
    ])

static func A4_defect(epsilon: float) -> PackedFloat32Array:
    # 缺陷 A₄：左上角减去 ε
    return PackedFloat32Array([
        2.0 - epsilon, -1.0,        0.0,        0.0,
       -1.0,           2.0,       -1.0,        0.0,
        0.0,          -1.0,        2.0,       -1.0,
        0.0,           0.0,       -1.0,        2.0
    ])
```

### 2.2 元素嘉当矩阵

$$\mathcal{C}_{\text{element}} = \left(\bigoplus_{i=1}^{Z} C_p\right) \oplus \left(\bigoplus_{j=1}^{N} C_n(\epsilon_j)\right)$$

- 质子扇区：Z 个 A₄ 块对角（每个 4×4）
- 中子扇区：N 个缺陷 A₄ 块对角（每个 4×4）
- 总维度：4(Z + N) × 4(Z + N)

```gdscript
static func element_cartan(Z: int, N: int, epsilon: float) -> PackedFloat32Array:
    var size = 4 * (Z + N)
    var mat = PackedFloat32Array()
    mat.resize(size * size)
    mat.fill(0.0)

    # 质子扇区：Z 个 A₄ 块
    for i in range(Z):
        _place_block(mat, size, A4(), 4 * i)

    # 中子扇区：N 个缺陷 A₄ 块
    for j in range(N):
        _place_block(mat, size, A4_defect(epsilon), 4 * (Z + j))

    return mat

static func _place_block(mat: PackedFloat32Array, size: int,
                          block: PackedFloat32Array, offset: int):
    var bs = 4  # 块大小
    for i in range(bs):
        for j in range(bs):
            mat[(offset + i) * size + (offset + j)] = block[i * bs + j]
```

### 2.3 分子超级嘉当矩阵

$$\mathcal{C}_{\text{mol}} = \bigoplus_{k=1}^{N_{\text{atom}}} \mathcal{C}_{\text{element}}^{(k)} + \sum_{\langle i,j \rangle} T_{ij}$$

- 第一项：各元素嘉当矩阵的块对角叠加
- 第二项：键耦合修正 T_ij

```gdscript
static func molecule_cartan(atoms: Array, bonds: Array) -> PackedFloat32Array:
    # 1. 计算每个元素的嘉当矩阵
    var element_mats = []
    var total_size = 0
    for atom in atoms:
        var Z = atom.atomic_number
        var N = atom.neutron_count
        var eps = atom.neutron_defect
        var mat = element_cartan(Z, N, eps)
        var sz = 4 * (Z + N)
        element_mats.append({"mat": mat, "size": sz})
        total_size += sz

    # 2. 块对角组装
    var mol_mat = PackedFloat32Array()
    mol_mat.resize(total_size * total_size)
    mol_mat.fill(0.0)

    var offset = 0
    for em in element_mats:
        _place_block_large(mol_mat, total_size, em.mat, em.size, offset)
        offset += em.size

    # 3. 添加键耦合修正
    for bond in bonds:
        var T = bond_coupling(bond, atoms, element_mats, total_size)
        _add_matrix(mol_mat, T, total_size)

    return mol_mat
```

### 2.4 键耦合修正 T_ij

键耦合描述两个原子间的因果相互作用：

$$T_{ij} = t_{ij} \cdot \hat{e}_{\alpha}^{(i)} \otimes \hat{e}_{\beta}^{(j)}$$

其中 t_ij 为耦合强度，α、β 为通道指标。

```gdscript
static func bond_coupling(bond: Dictionary, atoms: Array,
                           element_mats: Array, total_size: int) -> PackedFloat32Array:
    var T = PackedFloat32Array()
    T.resize(total_size * total_size)
    T.fill(0.0)

    var i = bond.a
    var j = bond.b
    var t_ij = compute_coupling_strength(atoms[i], atoms[j], bond)

    # 计算偏移
    var offset_i = 0
    for k in range(i):
        offset_i += element_mats[k].size
    var offset_j = 0
    for k in range(j):
        offset_j += element_mats[k].size

    # 在 (i 通道 α, j 通道 β) 位置填入 t_ij
    # 简化：仅在主通道（α=β=0）耦合
    T[(offset_i) * total_size + (offset_j)] = t_ij
    T[(offset_j) * total_size + (offset_i)] = t_ij  # 对称

    return T

static func compute_coupling_strength(a: Dictionary, b: Dictionary,
                                        bond: Dictionary) -> float:
    # 耦合强度 = f(键长, 元素类型, 键级)
    var r = a.position.distance_to(b.position)
    var r0 = _equilibrium_bond_length(a.symbol, b.symbol)
    var order = bond.get("order", 1)

    # 指数衰减耦合
    var t0 = 1.0  # 基准耦合
    var decay = 1.5  # 衰减常数 (Å⁻¹)
    var coupling = t0 * order * exp(-decay * abs(r - r0))

    # 电负性修正
    var chi_a = ElementDatabase.get_element(a.symbol).electronegativity
    var chi_b = ElementDatabase.get_element(b.symbol).electronegativity
    var polarity = abs(chi_a - chi_b) / 4.0  # 归一化极性
    coupling *= (1.0 - 0.3 * polarity)  # 极性键减弱超导耦合

    return coupling

static func _equilibrium_bond_length(sym_a: String, sym_b: String) -> float:
    var ra = ElementDatabase.get_element(sym_a).covalent_radius_pm / 100.0  # Å
    var rb = ElementDatabase.get_element(sym_b).covalent_radius_pm / 100.0
    return ra + rb
```

## 3. Step 2: 谱分析

### 3.1 本征值计算

对 4×4 A₄ 矩阵，本征值有解析解：

$$\lambda_k = 2 - 2\cos\frac{k\pi}{5}, \quad k = 1, 2, 3, 4$$

```gdscript
static func A4_eigenvalues() -> PackedFloat32Array:
    var vals = PackedFloat32Array()
    for k in range(1, 5):
        vals.append(2.0 - 2.0 * cos(k * PI / 5.0))
    return vals
    # 结果：[0.382, 1.382, 2.618, 3.618]
```

### 3.2 大矩阵数值本征值

对分子超级嘉当矩阵（可能很大），使用数值方法：

```gdscript
# eigenvalue_solver.gd
class_name EigenvalueSolver
extends RefCounted

static func eigenvalues_jacobi(mat: PackedFloat32Array, size: int,
                                max_iter: int = 100, tol: float = 1e-8) -> PackedFloat32Array:
    # Jacobi 旋转法（对称矩阵）
    var a = mat.duplicate()
    var n = size

    for iteration in range(max_iter):
        # 找最大非对角元素
        var p = 0
        var q = 1
        var max_val = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                var val = abs(a[i * n + j])
                if val > max_val:
                    max_val = val
                    p = i
                    q = j

        if max_val < tol:
            break

        # Jacobi 旋转
        var app = a[p * n + p]
        var aqq = a[q * n + q]
        var apq = a[p * n + q]
        var theta = 0.5 * atan2(2.0 * apq, app - aqq)
        var c = cos(theta)
        var s = sin(theta)

        for i in range(n):
            var aip = a[i * n + p]
            var aiq = a[i * n + q]
            a[i * n + p] = c * aip + s * aiq
            a[i * n + q] = -s * aip + c * aiq
        for j in range(n):
            var apj = a[p * n + j]
            var aqj = a[q * n + j]
            a[p * n + j] = c * apj + s * aqj
            a[q * n + j] = -s * apj + c * aqj

    # 提取对角元素
    var eigenvals = PackedFloat32Array()
    for i in range(n):
        eigenvals.append(a[i * n + i])
    eigenvals.sort()
    return eigenvals
```

### 3.3 谱间隙

```gdscript
static func spectral_gap(eigenvalues: PackedFloat32Array) -> float:
    if eigenvalues.size() < 2:
        return 0.0
    var sorted = eigenvalues.duplicate()
    sorted.sort()
    # 最小正本征值
    for v in sorted:
        if v > 1e-10:
            return v
    return 0.0
```

### 3.4 态密度

$$\rho(\lambda) = \sum_k \delta(\lambda - \lambda_k)$$

实际计算用高斯展宽：

$$\rho(\lambda) = \frac{1}{N\sqrt{2\pi}\sigma} \sum_k \exp\left(-\frac{(\lambda - \lambda_k)^2}{2\sigma^2}\right)$$

```gdscript
static func density_of_states(eigenvalues: PackedFloat32Array,
                                energy: float, sigma: float = 0.1) -> float:
    var n = eigenvalues.size()
    var dos = 0.0
    for ev in eigenvalues:
        dos += exp(-pow(energy - ev, 2) / (2.0 * sigma * sigma))
    dos /= n * sqrt(2.0 * PI) * sigma
    return dos

static func dos_at_fermi(eigenvalues: PackedFloat32Array,
                          fermi_level: float) -> float:
    return density_of_states(eigenvalues, fermi_level, 0.1)
```

## 4. Step 3: 因果截断频率

### 4.1 因果截断频率公式

CQM 理论中，因果截断频率由谱量子 C 和质子质量决定：

$$\omega_{\text{causal}} = \frac{C \cdot c}{l_P} \cdot f(\lambda_1)$$

其中：
- C = 0.02309570897（谱量子）
- c = 光速
- l_P = 普朗克长度
- f(λ₁) = 谱间隙修正函数

```gdscript
# cqm_constants.gd
class_name CQMConstants
extends RefCounted

# CQM 核心常数（形式化验证值）
const SPECTRAL_QUANTUM_C = 0.02309570897   # ξ'(1)/ξ(1)
const FIRST_COUPLING_E1 = 200.04045483     # 1/4 + γ₁²
const MATHIEU_CRITICAL = 1.316022911       # b₁(q) = 2q
const DYNKIN_INDEX = 5.0 / 3.0             # SU(5)
const A4_SPECTRAL_GAP = 0.38196601125      # (3-√5)/2
const BCS_PREFACTOR = 1.1339               # 2e^γ/π
const UNIVERSAL_GAP_RATIO = 3.5278         # 2πe^{-γ}

# 物理常数
const SPEED_OF_LIGHT = 2.998e8             # m/s
const PLANCK_LENGTH = 1.616e-35            # m
const PLANCK_CONSTANT = 6.626e-34          # J·s
const HBAR = 1.055e-34                     # J·s
const BOLTZMANN = 1.381e-23                # J/K
const PROTON_MASS = 1.673e-27              # kg
const ELECTRON_MASS = 9.109e-31            # kg
const EV_TO_KELVIN = 11604.5               # 1 eV = 11604.5 K
```

### 4.2 因果截断频率计算

```gdscript
# cqm_calculator.gd
class_name CQMCalculator
extends Node

static func causal_cutoff_frequency(spectral_gap: float) -> float:
    # ω_causal = (C · c / l_P) · f(λ₁)
    var base = CQMConstants.SPECTRAL_QUANTUM_C * CQMConstants.SPEED_OF_LIGHT / CQMConstants.PLANCK_LENGTH
    # 谱间隙修正：f(λ₁) = λ₁ / λ₁(A₄)
    var correction = spectral_gap / CQMConstants.A4_SPECTRAL_GAP
    return base * correction

static func causal_cutoff_energy(spectral_gap: float) -> float:
    # ħω_causal (J)
    var omega = causal_cutoff_frequency(spectral_gap)
    return CQMConstants.HBAR * omega

static func causal_cutoff_temperature(spectral_gap: float) -> float:
    # ħω_causal / k_B (K)
    var energy = causal_cutoff_energy(spectral_gap)
    return energy / CQMConstants.BOLTZMANN
```

## 5. Step 4: 涌现积分估算

### 5.1 涌现积分公式

$$\psi(\mathbf{r}, T) = \int_{\text{BZ}} d^3k \; \mathcal{D}_{\text{lattice}} \cdot \mathcal{P}_{\text{electron}} \cdot \mathcal{C}_{\text{triple}} \cdot \mathcal{K}_{\text{causal}} \cdot e^{-\Gamma_\phi|\tau|}$$

各因子含义：

| 因子 | 含义 | 计算方式 |
|:---|:---|:---|
| $\mathcal{D}_{\text{lattice}}$ | 晶格传播子 | 由嘉当矩阵本征值构造 |
| $\mathcal{P}_{\text{electron}}$ | 电子传播子 | BCS 形式 |
| $\mathcal{C}_{\text{triple}}$ | 三重因果修正 | CQM 高阶修正 |
| $\mathcal{K}_{\text{causal}}$ | 因果核 | 指数衰减 |
| $e^{-\Gamma_\phi|\tau|}$ | 因果阻尼 | Γ_φ 为因果衰减率 |

### 5.2 简化估算（快速模式）

对设计器用途，采用简化估算而非完整数值积分：

```gdscript
static func emergence_integral_fast(
    eigenvalues: PackedFloat32Array,
    spectral_gap: float,
    temperature: float,
    dos_fermi: float
) -> Dictionary:
    # 1. 因果截断温度
    var T_causal = causal_cutoff_temperature(spectral_gap)

    # 2. BCS 指数因子
    var coupling = estimate_coupling_constant(eigenvalues, dos_fermi)
    var bcs_exp = exp(-1.0 / coupling) if coupling > 0 else 0.0

    # 3. 涌现积分主项
    var psi_0 = CQMConstants.BCS_PREFACTOR * T_causal * bcs_exp

    # 4. 三重因果修正（一阶）
    var triple_correction = 1.0 + CQMConstants.SPECTRAL_QUANTUM_C * spectral_gap

    # 5. 温度依赖
    var tau = 1.0 - temperature / T_causal if T_causal > 0 else 0.0
    var temp_factor = sqrt(max(0.0, tau))

    var psi = psi_0 * triple_correction * temp_factor

    return {
        "psi": psi,
        "T_causal": T_causal,
        "coupling": coupling,
        "bcs_exponent": bcs_exp,
        "triple_correction": triple_correction,
        "temperature_factor": temp_factor
    }

static func estimate_coupling_constant(eigenvalues: PackedFloat32Array,
                                         dos_fermi: float) -> float:
    # N(0) · V_0 估算
    # V_0 由谱间隙决定：小谱间隙 → 强耦合
    var gap = spectral_gap(eigenvalues)
    if gap <= 0:
        return 0.0
    var V0 = 1.0 / (1.0 + gap / CQMConstants.A4_SPECTRAL_GAP)
    return dos_fermi * V0
```

### 5.3 完整数值积分（高精度模式）

```gdscript
static func emergence_integral_full(
    cartan_matrix: PackedFloat32Array,
    matrix_size: int,
    temperature: float,
    k_points: int = 32
) -> Dictionary:
    # 在布里渊区采样 k 点
    var psi = Vector3.ZERO
    var total_weight = 0.0

    for i in range(k_points):
        for j in range(k_points):
            for k in range(k_points):
                var kx = (float(i) + 0.5) / k_points - 0.5
                var ky = (float(j) + 0.5) / k_points - 0.5
                var kz = (float(k) + 0.5) / k_points - 0.5
                var k_vec = Vector3(kx, ky, kz)

                var weight = 1.0 / (k_points ** 3)
                var integrand = _integrand(k_vec, cartan_matrix, matrix_size, temperature)
                psi += integrand * weight
                total_weight += weight

    return {"psi": psi, "k_points": k_points ** 3, "weight": total_weight}

static func _integrand(k: Vector3, mat: PackedFloat32Array,
                        size: int, T: float) -> Vector3:
    # D_lattice: 由嘉当矩阵构造的晶格传播子
    var D = _lattice_propagator(k, mat, size)

    # P_electron: BCS 电子传播子
    var P = _electron_propagator(k, T)

    # K_causal: 因果核
    var K = _causal_kernel(k)

    return D * P * K
```

## 6. Step 5: 临界温度 T_c 估算

### 6.1 BCS 公式（CQM 修正）

$$k_B T_c = \frac{2e^\gamma}{\pi} \cdot \hbar\omega_{\text{causal}} \cdot \exp\left(-\frac{1}{N(0)\cdot V_0}\right)$$

CQM 修正：
- $\omega_{\text{causal}}$ 由因果截断决定（非 Debye 频率）
- $V_0$ 由嘉当矩阵谱间隙决定
- 多分量修正因子

```gdscript
static func estimate_tc(
    eigenvalues: PackedFloat32Array,
    spectral_gap: float,
    dos_fermi: float,
    pressure: float = 0.0
) -> Dictionary:
    # 1. 因果截断温度
    var T_causal = causal_cutoff_temperature(spectral_gap)

    # 2. 耦合常数 N(0)·V_0
    var coupling = estimate_coupling_constant(eigenvalues, dos_fermi)
    if coupling <= 0:
        return {"tc": 0.0, "reason": "非正耦合"}

    # 3. BCS 指数因子
    var bcs_exp = exp(-1.0 / coupling)

    # 4. T_c 主项
    var tc = CQMConstants.BCS_PREFACTOR * T_causal * bcs_exp

    # 5. 压力修正（高压增强氢化物 T_c）
    var pressure_factor = _pressure_correction(pressure)
    tc *= pressure_factor

    # 6. 多分量修正（A₄ 四通道）
    var multicomponent_factor = _multicomponent_correction(eigenvalues)
    tc *= multicomponent_factor

    return {
        "tc": tc,
        "T_causal": T_causal,
        "coupling": coupling,
        "bcs_exp": bcs_exp,
        "pressure_factor": pressure_factor,
        "multicomponent_factor": multicomponent_factor
    }

static func _pressure_correction(pressure_gpa: float) -> float:
    # 高压下氢化物 T_c 增强（经验公式）
    # LaH10 在 170 GPa 达到 203 K
    if pressure_gpa <= 0:
        return 1.0
    return 1.0 + 0.3 * log10(1.0 + pressure_gpa / 50.0)

static func _multicomponent_correction(eigenvalues: PackedFloat32Array) -> float:
    # A₄ 有 4 个通道，多分量增强
    # 修正因子 = (Σ_k λ_k^2) / (4 · λ₁^2)
    if eigenvalues.size() < 4:
        return 1.0
    var sum_sq = 0.0
    for ev in eigenvalues:
        sum_sq += ev * ev
    var gap = spectral_gap(eigenvalues)
    if gap <= 0:
        return 1.0
    return sum_sq / (4.0 * gap * gap)
```

### 6.2 强耦合修正

对强耦合情况（N(0)·V_0 > 1），BCCS 弱耦合公式不适用，采用 McMillan 公式：

$$T_c = \frac{\omega_{\text{causal}}}{1.2} \cdot \exp\left(-\frac{1.04(1+\lambda)}{\lambda - \mu^*(1+0.62\lambda)}\right)$$

```gdscript
static func estimate_tc_strong_coupling(
    omega_causal: float,
    coupling_lambda: float,
    mu_star: float = 0.1
) -> float:
    if coupling_lambda <= mu_star:
        return 0.0
    var exponent = -1.04 * (1.0 + coupling_lambda) / \
                   (coupling_lambda - mu_star * (1.0 + 0.62 * coupling_lambda))
    return omega_causal / 1.2 * exp(exponent)
```

## 7. Step 6: 多分量序参量

### 7.1 A₄ 根系分解

A₄ 有 4 个通道（对应 4 个本征值），序参量为 4 分量向量：

$$\vec{\Delta} = (\Delta_1, \Delta_2, \Delta_3, \Delta_4)$$

各分量由本征值和耦合决定：

$$\Delta_k = \Delta_0 \cdot \sqrt{\lambda_k} \cdot e^{i\phi_k}$$

```gdscript
static func order_parameters(
    eigenvalues: PackedFloat32Array,
    tc: float,
    temperature: float
) -> Array:
    if tc <= 0 or temperature >= tc:
        return []

    # BCS 能隙
    var delta_0 = CQMConstants.UNIVERSAL_GAP_RATIO * CQMConstants.BOLTZMANN * tc / 2.0
    # 温度修正
    var temp_ratio = temperature / tc
    var temp_correction = sqrt(max(0.0, 1.0 - temp_ratio))
    delta_0 *= temp_correction

    # 各通道序参量
    var params = []
    for k in range(min(4, eigenvalues.size())):
        var lambda_k = eigenvalues[k]
        var amplitude = delta_0 * sqrt(lambda_k)
        # 相位（简化：均匀分布）
        var phase = k * PI / 4.0
        params.append({
            "channel": k + 1,
            "eigenvalue": lambda_k,
            "amplitude": amplitude,
            "phase": phase,
            "real": amplitude * cos(phase),
            "imag": amplitude * sin(phase)
        })

    return params
```

### 7.2 分步相变

多分量系统可能发生分步相变：

```
温度降低
  │
  ├─ T > T_c₁: 常规态，Δ = 0
  │
  ├─ T_c₂ < T < T_c₁: 通道 1 凝聚，Δ₁ ≠ 0
  │
  ├─ T_c₃ < T < T_c₂: 通道 1,2 凝聚
  │
  ├─ T_c₄ < T < T_c₃: 通道 1,2,3 凝聚
  │
  └─ T < T_c₄: 全部通道凝聚，完全超导态
```

```gdscript
static func stepwise_transition(
    eigenvalues: PackedFloat32Array,
    tc: float
) -> Array:
    # 各通道的凝聚温度
    var transitions = []
    var sorted_ev = eigenvalues.duplicate()
    sorted_ev.sort()

    for k in range(min(4, sorted_ev.size())):
        var lambda_k = sorted_ev[k]
        var gap = spectral_gap(eigenvalues)
        if gap <= 0:
            continue
        # 通道 k 的 T_c ∝ √λ_k
        var tc_k = tc * sqrt(lambda_k / sorted_ev[-1])
        transitions.append({
            "channel": k + 1,
            "tc_channel": tc_k,
            "eigenvalue": lambda_k
        })

    transitions.sort_custom(func(a, b): return a.tc_channel > b.tc_channel)
    return transitions
```

## 8. Step 7: 超导判定

### 8.1 判定准则

| 条件 | 判定 | 说明 |
|:---|:---|:---|
| T_c > T_target 且置信度高 | `superconducting` | 超导 |
| T_c ≈ T_target（±20%） | `borderline` | 边界 |
| T_c < T_target 或 T_c = 0 | `normal` | 常规态 |
| 数据不足 | `insufficient` | 无法判定 |

### 8.2 置信度评估

```gdscript
static func confidence_score(
    eigenvalues: PackedFloat32Array,
    coupling: float,
    spectral_gap: float,
    atom_count: int
) -> float:
    var score = 0.0

    # 1. 耦合强度置信（0-0.3）
    if coupling > 0:
        score += 0.3 * min(1.0, coupling)

    # 2. 谱间隙置信（0-0.3）
    var gap_ratio = spectral_gap / CQMConstants.A4_SPECTRAL_GAP
    score += 0.3 * min(1.0, gap_ratio)

    # 3. 原子数置信（0-0.2）
    # 太少不可靠，太多计算量大
    var atom_factor = 1.0 - abs(atom_count - 10) / 20.0
    score += 0.2 * clamp(atom_factor, 0.0, 1.0)

    # 4. 本征值分布置信（0-0.2）
    if eigenvalues.size() >= 4:
        var ev = eigenvalues
        var spread = (ev[-1] - ev[0]) / ev[-1] if ev[-1] > 0 else 0
        score += 0.2 * min(1.0, spread)

    return clamp(score, 0.0, 1.0)
```

### 8.3 综合判定

```gdscript
static func evaluate_molecule(atoms: Array, bonds: Array,
                                params: Dictionary = {}) -> Dictionary:
    var temperature = params.get("temperature", 4.2)
    var pressure = params.get("pressure", 0.0)
    var precision = params.get("precision", "normal")

    # Step 1: 嘉当矩阵组装
    var cartan = CQMMatrix.molecule_cartan(atoms, bonds)
    var mat_size = _infer_size(atoms)

    # Step 2: 谱分析
    var eigenvalues = EigenvalueSolver.eigenvalues_jacobi(cartan, mat_size)
    var gap = spectral_gap(eigenvalues)
    var fermi = _estimate_fermi_level(eigenvalues)
    var dos_f = density_of_states(eigenvalues, fermi)

    # Step 3: 因果截断频率
    var omega_c = causal_cutoff_frequency(gap)
    var T_causal = causal_cutoff_temperature(gap)

    # Step 4: 涌现积分
    var emergence
    if precision == "fast":
        emergence = emergence_integral_fast(eigenvalues, gap, temperature, dos_f)
    else:
        emergence = emergence_integral_fast(eigenvalues, gap, temperature, dos_f)

    # Step 5: T_c 估算
    var tc_result = estimate_tc(eigenvalues, gap, dos_f, pressure)
    var tc = tc_result.tc

    # Step 6: 多分量序参量
    var order_params = order_parameters(eigenvalues, tc, temperature)
    var transitions = stepwise_transition(eigenvalues, tc)

    # Step 7: 判定
    var coupling = tc_result.coupling
    var confidence = confidence_score(eigenvalues, coupling, gap, atoms.size())
    var verdict = _make_verdict(tc, temperature, confidence)

    return {
        "verdict": verdict,
        "tc_estimate": tc,
        "confidence": confidence,
        "cartan_matrix": cartan,
        "matrix_size": mat_size,
        "eigenvalues": eigenvalues,
        "spectral_gap": gap,
        "fermi_level": fermi,
        "dos_fermi": dos_f,
        "causal_cutoff_freq": omega_c,
        "causal_cutoff_temp": T_causal,
        "emergence": emergence,
        "order_parameters": order_params,
        "stepwise_transitions": transitions,
        "tc_breakdown": tc_result,
        "atom_count": atoms.size(),
        "bond_count": bonds.size()
    }

static func _make_verdict(tc: float, target_temp: float,
                            confidence: float) -> String:
    if confidence < 0.2:
        return "insufficient"
    if tc <= 0:
        return "normal"
    if tc > target_temp * 1.2:
        return "superconducting"
    if tc > target_temp * 0.8:
        return "borderline"
    return "normal"
```

## 9. BCS 退化路径

### 9.1 弱耦合极限

CQM 理论在弱耦合极限退化为标准 BCS：

$$T_c \xrightarrow{V_0 \to 0} \frac{2e^\gamma}{\pi} \cdot \omega_D \cdot \exp\left(-\frac{1}{N(0)V_0}\right)$$

```gdscript
static func bcs_degradation_path(eigenvalues: PackedFloat32Array,
                                   coupling: float) -> Dictionary:
    # 从 CQM 到 BCS 的退化参数
    var path = {
        "cqm_full": 0.0,
        "cqm_first_order": 0.0,
        "bcs_standard": 0.0
    }

    var gap = spectral_gap(eigenvalues)
    var T_causal = causal_cutoff_temperature(gap)

    # 1. CQM 完整（含三重修正）
    var triple = 1.0 + CQMConstants.SPECTRAL_QUANTUM_C * gap
    path.cqm_full = CQMConstants.BCS_PREFACTOR * T_causal * exp(-1.0/coupling) * triple

    # 2. CQM 一阶（仅因果截断修正）
    path.cqm_first_order = CQMConstants.BCS_PREFACTOR * T_causal * exp(-1.0/coupling)

    # 3. 标准 BCS（Debye 频率替代因果截断）
    var T_debye = 300.0  # 典型 Debye 温度 (K)
    path.bcs_standard = CQMConstants.BCS_PREFACTOR * T_debye * exp(-1.0/coupling)

    return path
```

### 10.2 降维策略

对大分子，采用以下降维：

```gdscript
static func reduce_dimension(cartan: PackedFloat32Array, size: int,
                               target_size: int = 64) -> Dictionary:
    if size <= target_size:
        return {"matrix": cartan, "size": size}

    # 1. 计算本征值
    var eigenvalues = EigenvalueSolver.eigenvalues_jacobi(cartan, size)

    # 2. 保留最大 target_size 个本征值对应的子空间
    # （投影到主成分）
    var reduced = PackedFloat32Array()
    reduced.resize(target_size * target_size)
    reduced.fill(0.0)
    for i in range(target_size):
        reduced[i * target_size + i] = eigenvalues[i + (size - target_size)]

    return {"matrix": reduced, "size": target_size, "original_size": size}
```

func evaluate_cached(atoms: Array, bonds: Array) -> Dictionary:
