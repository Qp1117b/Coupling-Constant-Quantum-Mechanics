## 1. 总体架构

### 1.1 五层双轨架构

超导材料设计从原子到宏观实体分为五层，支持模式设计（经验规则）与自由设计（物理约束）双轨工作流：

```
┌─────────────────────────────────────────────────────────┐
│ L0 化学式 (全局锚点) │
│ 由工作区原子自动推导，驱动模式筛选 │
├─────────────────────────────────────────────────────────┤
│ 上屏 (分子/原子设计) │ 下屏 (画笔=L4晶胞分布) │
│ L1 结合方式 (骨架拓扑) │ L3 晶胞 (周期性堆积) │
│ L2 相对位置 (几何参数) │ L4 晶胞分布(晶粒) │
│ 模式库: 保守子结构/可变模块 │ 单晶/多晶/薄膜/粉末 │
│ 键长/键角/构象 │ 画笔绘制→Regge单元 │
│ 右键结合连接→自动匹配模式 │ →嘉当矩阵→FG退相干场 │
├─────────────────────────────────────────────────────────┤
│ 模式引擎 (PatternEngine) │
│ 双轨: 模式设计(经验规则挂载) / 自由设计(物理硬约束) │
│ 4家族: 铜氧化物/铁基/氢化物/BCS │
└─────────────────────────────────────────────────────────┘
```

## 2. 核心节点设计

### 2.1 Atom3D —— 原子节点

```gdscript
# atom.gd — 原子节点脚本
class_name Atom3D
extends MeshInstance3D

# 元素信息
@export var element_symbol: String = "H" # 元素符号
@export var atomic_number: int = 1 # 原子序数
@export var isotope_mass: int = 1 # 同位素质量数
@export var neutron_count: int = 0 # 中子数

# 3D 属性
@export var atom_radius: float = 0.3 # 原子半径（显示）
@export var color: Color = Color.WHITE # 元素颜色

# CQM 属性
var cartan_matrix: Matrix # 元素嘉当矩阵
var neutron_defect: float = 0.0 # 中子缺陷参数 ε
var eigenvalues: Array[float] = [] # A₄ 本征值

# 信号
signal atom_placed(atom: Atom3D)
signal atom_selected(atom: Atom3D)
signal atom_moved(atom: Atom3D, new_pos: Vector3)

func _ready():
 # 加载元素数据
 _load_element_data()
 # 构建 A₄ 嘉当矩阵
 _build_cartan_matrix()
 # 设置材质
 _setup_material()

func _build_cartan_matrix():
 # 质子扇区：Z 个 A₄ 块对角
 # 中子扇区：N 个缺陷 A₄ 块对角
 cartan_matrix = CQMCalculator.build_element_cartan(
 atomic_number, neutron_count, neutron_defect
 )
 eigenvalues = cartan_matrix.eigenvalues()
```

### 2.2 Bond3D —— 化学键节点

```gdscript
# bond.gd — 化学键节点
class_name Bond3D
extends MeshInstance3D

@export var atom_a: Atom3D # 键端原子 A
@export var atom_b: Atom3D # 键端原子 B
@export var bond_order: int = 1 # 键级（1=单, 2=双, 3=三）
@export var bond_type: String = "covalent" # 键类型

# CQM 属性
var coupling_strength: float = 0.0 # 因果耦合 t_ij
var bond_length: float = 0.0 # 键长

func _process(delta):
 # 实时更新键的几何（连接两个原子）
 _update_geometry()

func _update_geometry():
 var pos_a = atom_a.global_position
 var pos_b = atom_b.global_position
 var midpoint = (pos_a + pos_b) / 2.0
 var direction = (pos_b - pos_a).normalized()
 bond_length = pos_a.distance_to(pos_b)

 # 更新位置和朝向
 global_position = midpoint
 look_at(pos_b, Vector3.UP)

 # 更新缩放（键长）
 scale.z = bond_length

 # 计算因果耦合强度
 coupling_strength = CQMCalculator.compute_coupling(
 atom_a.element_symbol, atom_b.element_symbol, bond_length
 )
```

### 2.3 MoleculeWorkspace —— 分子构建空间

```gdscript
# molecule_workspace.gd
class_name MoleculeWorkspace
extends Node3D

var atoms: Array[Atom3D] = []
var bonds: Array[Bond3D] = []
var selected_atom: Atom3D = null
var pending_bond_from: Atom3D = null # 待连接键的起始原子

# 模式
enum Mode { PLACE, SELECT, BOND, DELETE, MEASURE }
var current_mode: Mode = Mode.PLACE

signal molecule_changed()
signal atom_added(atom: Atom3D)
signal bond_added(bond: Bond3D)

func add_atom(symbol: String, isotope: int, position: Vector3) -> Atom3D:
 var atom = Atom3D.new()
 atom.element_symbol = symbol
 atom.isotope_mass = isotope
 atom.position = position
 $Atoms.add_child(atom)
 atoms.append(atom)
 atom_placed.emit(atom)
 molecule_changed.emit()
 return atom

func add_bond(a: Atom3D, b: Atom3D, order: int = 1) -> Bond3D:
 var bond = Bond3D.new()
 bond.atom_a = a
 bond.atom_b = b
 bond.bond_order = order
 $Bonds.add_child(bond)
 bonds.append(bond)
 bond_added.emit(bond)
 molecule_changed.emit()
 return bond

func get_molecule_cartan() -> Matrix:
 # 组装分子有效超级嘉当矩阵
 # C_mol = ⊕_k C_el(k) + Σ_{i<j} T_ij
 return CQMCalculator.build_molecule_cartan(atoms, bonds)
```

## 3. 交互系统设计

### 3.1 积木拼接交互流程

```
用户操作 系统响应
──────── ────────
1. 从元素周期表选择元素 → 高亮该元素，进入"放置模式"
2. 在 3D 空间点击 → 在该位置创建原子节点
3. 拖拽原子 → 实时更新位置，更新键几何
4. 选择"键连接"模式 → 进入键连接模式
5. 点击原子 A → 标记为键起始
6. 点击原子 B → 创建 A-B 键
7. 选择同位素 → 更新中子数，重建嘉当矩阵
8. 点击"计算" → 触发 CQM 超导判定
```

### 3.2 3D 拾取与放置

```gdscript
# interaction_controller.gd
extends Node3D

@export var camera: Camera3D
@export var workspace: MoleculeWorkspace
@export var ray_length: float = 1000.0

var current_element: String = "H"
var current_isotope: int = 1

func _unhandled_input(event):
 if event is InputEventMouseButton:
 if event.pressed and event.button_index == MOUSE_BUTTON_LEFT:
 _handle_left_click(event.position)
 elif event is InputEventMouseMotion:
 _handle_mouse_move(event.relative)

func _handle_left_click(screen_pos: Vector2):
 var ray = _screen_to_ray(screen_pos)

 match workspace.current_mode:
 Mode.PLACE:
 var pos = _ray_to_ground(ray)
 workspace.add_atom(current_element, current_isotope, pos)
 Mode.SELECT:
 var atom = _ray_to_atom(ray)
 if atom:
 workspace.select_atom(atom)
 Mode.BOND:
 var atom = _ray_to_atom(ray)
 if atom:
 _handle_bond_click(atom)
 Mode.DELETE:
 var node = _ray_to_node(ray)
 if node:
 workspace.delete_node(node)

func _screen_to_ray(screen_pos: Vector2) -> Dictionary:
 var from = camera.project_ray_origin(screen_pos)
 var dir = camera.project_ray_normal(screen_pos)
 return {"from": from, "dir": dir}

func _ray_to_atom(ray: Dictionary) -> Atom3D:
 var space = get_world_3d().direct_space_state
 var query = PhysicsRayQueryParameters3D.create(
 ray.from, ray.from + ray.dir * ray_length
 )
 var result = space.intersect_ray(query)
 if result and result.collider is Atom3D:
 return result.collider
 return null
```

### 3.3 相机轨道控制

```gdscript
# camera_orbit.gd
extends Node3D

@export var target: Node3D
@export var distance: float = 10.0
@export var min_distance: float = 2.0
@export var max_distance: float = 50.0
@export var rotate_speed: float = 0.005
@export var zoom_speed: float = 0.1

var yaw: float = 0.0
var pitch: float = -0.5
var _is_dragging: bool = false

func _unhandled_input(event):
 if event is InputEventMouseButton:
 if event.button_index == MOUSE_BUTTON_RIGHT:
 _is_dragging = event.pressed
 elif event.button_index == MOUSE_BUTTON_WHEEL_UP:
 distance = max(min_distance, distance - zoom_speed)
 elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
 distance = min(max_distance, distance + zoom_speed)
 elif event is InputEventMouseMotion and _is_dragging:
 yaw -= event.relative.x * rotate_speed
 pitch = clamp(pitch - event.relative.y * rotate_speed, -1.5, 1.5)

func _process(delta):
 if target:
 var offset = Vector3(
 distance * cos(pitch) * sin(yaw),
 distance * sin(pitch),
 distance * cos(pitch) * cos(yaw)
 )
 global_position = target.global_position + offset
 look_at(target.global_position, Vector3.UP)
```

### 3.4 吸附与对齐

为实现"积木"体验，原子放置时支持吸附：

```gdscript
# snapping.gd
extends Node

@export var grid_size: float = 1.0 # 网格大小
@export var snap_to_grid: bool = true # 网格吸附加
@export var snap_to_atom: bool = true # 原子吸附加
@export var atom_snap_radius: float = 0.5 # 原子吸附加范围
@export var snap_to_bond: bool = true # 键中点吸附加

func snap_position(pos: Vector3, existing_atoms: Array[Atom3D]) -> Vector3:
 var snapped = pos

 # 1. 网格吸附加
 if snap_to_grid:
 snapped = snapped.snapped(Vector3.ONE * grid_size)

 # 2. 原子吸附加（找最近的已有原子）
 if snap_to_atom:
 for atom in existing_atoms:
 if pos.distance_to(atom.position) < atom_snap_radius:
 # 吸附加到键长位置（而非重叠）
 var direction = (pos - atom.position).normalized()
 snapped = atom.position + direction * _get_bond_length(atom)
 break

 return snapped
```

```gdscript
# element_palette.gd
extends Control

const ELEMENTS_PER_ROW = 18
const PERIODS = 7

var element_buttons: Dictionary = {} # symbol -> Button

func _ready():
 _build_periodic_table()

func _build_periodic_table():
 var grid = $GridContainer
 grid.columns = ELEMENTS_PER_ROW

 for period in range(1, PERIODS + 1):
 for group in range(1, ELEMENTS_PER_ROW + 1):
 var element = ElementDatabase.get_element_by_position(period, group)
 if element:
 var btn = Button.new()
 btn.text = element.symbol
 btn.custom_minimum_size = Vector2(40, 40)
 btn.tooltip_text = "%s (Z=%d)" % [element.name, element.atomic_number]
 btn.modulate = element.color
 btn.pressed.connect(_on_element_selected.bind(element.symbol))
 grid.add_child(btn)
 element_buttons[element.symbol] = btn

func _on_element_selected(symbol: String):
 # 高亮选中元素
 for btn in element_buttons.values():
 btn.button_pressed = false
 element_buttons[symbol].button_pressed = true

 # 通知工作空间
 Events.element_selected.emit(symbol)
```
